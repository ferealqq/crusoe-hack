import numpy as np
import pandas as pd
from catboost import CatBoostClassifier
from lightgbm import LGBMClassifier
from sklearn.model_selection import KFold
from sklearn.metrics import log_loss

def fit(train: pd.DataFrame, target: str, seed: int) -> object:
    X = train.drop(columns=[target])
    y = train[target]
    
    # Define base models
    def get_catboost():
        return CatBoostClassifier(
            iterations=200,
            learning_rate=0.05,
            depth=6,
            loss_function='Logloss',
            random_seed=seed,
            verbose=False,
            thread_count=2
        )
    def get_lightgbm():
        return LGBMClassifier(
            n_estimators=300,
            learning_rate=0.03,
            num_leaves=31,
            subsample=0.8,
            colsample_bytree=0.8,
            random_state=seed,
            n_jobs=2
        )
    
    # Generate out-of-fold predictions
    n_splits = 5
    kf = KFold(n_splits=n_splits, shuffle=True, random_state=seed)
    oof_cb = np.zeros(len(X))
    oof_lgb = np.zeros(len(X))
    
    for fold, (train_idx, val_idx) in enumerate(kf.split(X)):
        X_tr, X_val = X.iloc[train_idx], X.iloc[val_idx]
        y_tr, y_val = y.iloc[train_idx], y.iloc[val_idx]
        
        # CatBoost
        cb = get_catboost()
        cb.fit(X_tr, y_tr)
        oof_cb[val_idx] = cb.predict_proba(X_val)[:, 1]
        
        # LightGBM
        lgb = get_lightgbm()
        lgb.fit(X_tr, y_tr)
        oof_lgb[val_idx] = lgb.predict_proba(X_val)[:, 1]
    
    # Compute CV log loss for each model
    loss_cb = log_loss(y, oof_cb)
    loss_lgb = log_loss(y, oof_lgb)
    
    # Compute weights: inverse loss, normalized
    w_cb = 1.0 / loss_cb
    w_lgb = 1.0 / loss_lgb
    total = w_cb + w_lgb
    w_cb /= total
    w_lgb /= total
    
    # Retrain base models on full data
    model_cb_full = get_catboost()
    model_cb_full.fit(X, y)
    model_lgb_full = get_lightgbm()
    model_lgb_full.fit(X, y)
    
    return (model_cb_full, model_lgb_full, w_cb, w_lgb)

def predict(model: object, X: pd.DataFrame) -> np.ndarray:
    model_cb_full, model_lgb_full, w_cb, w_lgb = model
    prob_cb = model_cb_full.predict_proba(X)[:, 1]
    prob_lgb = model_lgb_full.predict_proba(X)[:, 1]
    return w_cb * prob_cb + w_lgb * prob_lgb

import pandas as pd
from catboost import CatBoostClassifier

from sklearn.model_selection import KFold
from sklearn.metrics import log_loss

def fit(train: pd.DataFrame, target: str, seed: int) -> object:
    X = train.drop(columns=[target])
    y = train[target]
    
    # Define base models
    def get_catboost():
        return CatBoostClassifier(
            iterations=200,
            learning_rate=0.05,
            depth=6,
            loss_function='Logloss',
            random_seed=seed,
            verbose=False,
            thread_count=2
        )
    def get_lightgbm():
        return LGBMClassifier(
            n_estimators=300,
            learning_rate=0.03,
            num_leaves=31,
            subsample=0.8,
            colsample_bytree=0.8,
            random_state=seed,
            n_jobs=2
        )
    
    # Generate out-of-fold predictions
    n_splits = 5
    kf = KFold(n_splits=n_splits, shuffle=True, random_state=seed)
    oof_cb = np.zeros(len(X))
    oof_lgb = np.zeros(len(X))
    
    for fold, (train_idx, val_idx) in enumerate(kf.split(X)):
        X_tr, X_val = X.iloc[train_idx], X.iloc[val_idx]
        y_tr, y_val = y.iloc[train_idx], y.iloc[val_idx]
        
        # CatBoost
        cb = get_catboost()
        cb.fit(X_tr, y_tr)
        oof_cb[val_idx] = cb.predict_proba(X_val)[:, 1]
        
        # LightGBM
        lgb = get_lightgbm()
        lgb.fit(X_tr, y_tr)
        oof_lgb[val_idx] = lgb.predict_proba(X_val)[:, 1]
    
    # Compute CV log loss for each model
    loss_cb = log_loss(y, oof_cb)
    loss_lgb = log_loss(y, oof_lgb)
    
    # Compute weights: inverse loss, normalized
    w_cb = 1.0 / loss_cb
    w_lgb = 1.0 / loss_lgb
    total = w_cb + w_lgb
    w_cb /= total
    w_lgb /= total
    
    # Retrain base models on full data
    model_cb_full = get_catboost()
    model_cb_full.fit(X, y)
    model_lgb_full = get_lightgbm()
    model_lgb_full.fit(X, y)
    
    return (model_cb_full, model_lgb_full, w_cb, w_lgb)

def predict(model: object, X: pd.DataFrame) -> np.ndarray:
    model_cb_full, model_lgb_full, w_cb, w_lgb = model
    prob_cb = model_cb_full.predict_proba(X)[:, 1]
    prob_lgb = model_lgb_full.predict_proba(X)[:, 1]
    return w_cb * prob_cb + w_lgb * prob_lgb
